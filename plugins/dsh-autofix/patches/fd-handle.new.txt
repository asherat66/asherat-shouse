            const imgs = window.dshDesktop.drainDroppedImages()
            if (imgs.length > 0) {
              const actx = props.sessionsCtx ? props.sessionsCtx.scope(optsRef.current.sessionId) : null
              const cv = actx ? actx.get('conversation') : null
              if (cv && typeof cv.createDraftImages === 'function') {
                const files2 = imgs.map(function (it) {
                  const bin = atob(it.base64)
                  const bytes = new Uint8Array(bin.length)
                  for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i)
                  return new File([bytes], it.name, { type: it.mime })
                })
                const atts = cv.createDraftImages(files2)
                const inp = cv.input && cv.input.for ? cv.input.for(actx) : null
                if (inp && typeof inp.addImages === 'function' && inp.addImages(atts.map(function (x) { return x.id; }))) {

                  return
                }
              }
